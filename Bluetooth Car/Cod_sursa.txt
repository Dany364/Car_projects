#include "BluetoothSerial.h"

BluetoothSerial SerialBT;

int IN1 = 26;
int IN2 = 27;
int IN3 = 14;
int IN4 = 12;

#define BUZZER 33

void forward();
void back();
void left();
void right();
void stopCar();

void setup() {
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(BUZZER, OUTPUT);
  digitalWrite(BUZZER, LOW);

  stopCar();

  SerialBT.begin("ESP32_CAR");
}

void loop() {
  if (SerialBT.available()) {
    char c = SerialBT.read();

    if      (c == 'F') forward();
    else if (c == 'B') back();
    else if (c == 'L') left();
    else if (c == 'R') right();
    else if (c == 'S') stopCar();
    else if (c == 'H') digitalWrite(BUZZER, HIGH);
    else if (c == 'h') digitalWrite(BUZZER, LOW);
  }
}

void forward() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void back() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void left() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void right() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void stopCar() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}
