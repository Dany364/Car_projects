#define ECHO_PIN 11

#define IR_REAR_RIGHT  33

const int maximum_distance = 200;
const int obstacleDistance = 25;

// Viteza normală
const int motorSpeed = 130;

// Viteza normală în avoidance
const int avoidanceSpeed = 100;

// Viteza viraj manual
const int turnSpeed = 180;

// Viteza viraj AVOIDANCE
const int avoidanceTurnSpeed = 180;

const int reverseTime = 400;
const int turnTime = 500;

const int servoCenter = 115;
const int servoRight = 50;
const int servoLeft = 170;


// ==================== MOD ====================

enum RobotMode {
  MODE_MANUAL,
  MODE_AVOIDANCE
};

RobotMode currentMode = MODE_MANUAL;


// ==================== COMANDA ESP ====================

String espCommand = "";


// ==================== DISTANTA ====================

long readDistance() {

  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);

  digitalWrite(TRIG_PIN, LOW);

  unsigned long duration =
    pulseIn(ECHO_PIN, HIGH, 30000);

  if (duration == 0) {
    return 250;
  }

  long cm = duration / 58;

  if (cm <= 0) {
    cm = 250;
  }

  if (cm > maximum_distance) {
    cm = maximum_distance;
  }

  return cm;
}


// ==================== MOTOARE ====================

void motorStop() {

  analogWrite(ENA, 0);
  analogWrite(ENB, 0);

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}


void motorForward() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);

  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);
}


void motorBackward() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);

  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);
}


void motorLeft() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);

  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);
}


void motorRight() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);

  analogWrite(ENA, motorSpeed);
  analogWrite(ENB, motorSpeed);
}

// ==================== AVOIDANCE ====================

int lookRight() {

  servoPosition = servoRight;
  servo.write(servoRight);

  delay(500);

  int distance = readDistance();

  delay(100);

  servoPosition = servoCenter;
  servo.write(servoCenter);

  return distance;
}


int lookLeft() {

  servoPosition = servoLeft;
  servo.write(servoLeft);

  delay(500);

  int distance = readDistance();

  delay(100);

  servoPosition = servoCenter;
  servo.write(servoCenter);

  return distance;
}


void runAvoidance() {

  long distance = readDistance();


  // ==================================================
  // DRUM LIBER
  // FAȚĂ -> motorRight()
  // Viteza = 100
  // ==================================================

  if (distance > obstacleDistance) {

    motorRight();

    analogWrite(ENA, avoidanceSpeed);
    analogWrite(ENB, avoidanceSpeed);

    return;
  }


  // ==================================================
  // OBSTACOL DETECTAT
  // ==================================================

  motorStop();
  delay(200);


  // ==================================================
  // RETRAGERE
  // SPATE -> motorLeft()
  // Viteza = 100
  // ==================================================

  motorLeft();

  analogWrite(ENA, avoidanceSpeed);
  analogWrite(ENB, avoidanceSpeed);

  delay(reverseTime);

  motorStop();
  delay(200);


  // ==================================================
  // VERIFICĂM DREAPTA
  // ==================================================

  int distanceRight = lookRight();

  if (currentMode != MODE_AVOIDANCE) {

    motorStop();

    return;
  }


  // ==================================================
  // VERIFICĂM STÂNGA
  // ==================================================

  int distanceLeft = lookLeft();

  if (currentMode != MODE_AVOIDANCE) {

    motorStop();

    return;
  }


  // ==================================================
  // ALEGEM DIRECȚIA CU MAI MULT SPAȚIU
  //
  // STÂNGA / DREAPTA INVERSATE
  // ==================================================

  if (distanceLeft >= distanceRight) {

    // ==================================================
    // STÂNGA AVOIDANCE
    // -> motorBackward()
    //
    // Viteza = 180
    // ==================================================

    motorBackward();

    analogWrite(ENA, avoidanceTurnSpeed);
    analogWrite(ENB, avoidanceTurnSpeed);

    delay(turnTime);

  }

  else {

    // ==================================================
    // DREAPTA AVOIDANCE
    // -> motorForward()
    //
    // Viteza = 180
    // ==================================================

    motorForward();

    analogWrite(ENA, avoidanceTurnSpeed);
    analogWrite(ENB, avoidanceTurnSpeed);

    delay(turnTime);
  }


  // ==================================================
  // STOP DUPĂ VIRAJ
  // ==================================================

  motorStop();
  delay(150);


  servoPosition = servoCenter;
  servo.write(servoCenter);


  // ==================================================
  // CONTINUĂM ÎNAINTE
  // Viteza = 100
  // ==================================================

  motorRight();

  analogWrite(ENA, avoidanceSpeed);
  analogWrite(ENB, avoidanceSpeed);
}


// ==================== LEDURI ====================

void updateLEDs() {

  int ldrValue = analogRead(LDR_PIN);

  if (ldrValue < 250) {

    digitalWrite(LED_FRONT_LEFT, HIGH);
    digitalWrite(LED_FRONT_RIGHT, HIGH);

    digitalWrite(LED_SIDE_LEFT, HIGH);
    digitalWrite(LED_SIDE_RIGHT, HIGH);

    digitalWrite(LED_REAR_LEFT, HIGH);
    digitalWrite(LED_REAR_RIGHT, HIGH);

    return;
  }


  digitalWrite(LED_FRONT_LEFT, LOW);
  digitalWrite(LED_FRONT_RIGHT, LOW);

  digitalWrite(LED_SIDE_LEFT, LOW);
  digitalWrite(LED_SIDE_RIGHT, LOW);

  digitalWrite(LED_REAR_LEFT, LOW);
  digitalWrite(LED_REAR_RIGHT, LOW);


  if (digitalRead(IR_FRONT_LEFT) == LOW) {
    digitalWrite(LED_FRONT_LEFT, HIGH);
  }

  if (digitalRead(IR_FRONT_RIGHT) == LOW) {
    digitalWrite(LED_FRONT_RIGHT, HIGH);
  }

  if (digitalRead(IR_REAR_LEFT) == LOW) {
    digitalWrite(LED_REAR_LEFT, HIGH);
  }

  if (digitalRead(IR_REAR_RIGHT) == LOW) {
    digitalWrite(LED_REAR_RIGHT, HIGH);
  }
}


// ==================== DATE SENZORI ====================

void updateIRParkingBuzzer() {

  static unsigned long lastBeep = 0;
  static bool buzzerOn = false;

  bool detected =
    (digitalRead(IR_FRONT_LEFT) == LOW) ||
    (digitalRead(IR_FRONT_RIGHT) == LOW) ||
    (digitalRead(IR_REAR_LEFT) == LOW) ||
    (digitalRead(IR_REAR_RIGHT) == LOW);

  if (!detected) {

    noTone(BUZZER_PIN);
    buzzerOn = false;

    return;
  }

  const unsigned long interval = 180;
  unsigned long now = millis();

  if (now - lastBeep >= interval) {

    lastBeep = now;

    if (buzzerOn) {

      noTone(BUZZER_PIN);
      buzzerOn = false;

    } else {

      tone(BUZZER_PIN, 1200);
      buzzerOn = true;
    }
  }
}


// ==================== DATE SENZORI -> ESP32 ====================

void sendSensorData() {

  int FL = digitalRead(IR_FRONT_LEFT) == LOW ? 1 : 0;
  int FR = digitalRead(IR_FRONT_RIGHT) == LOW ? 1 : 0;
  int RL = digitalRead(IR_REAR_LEFT) == LOW ? 1 : 0;
  int RR = digitalRead(IR_REAR_RIGHT) == LOW ? 1 : 0;

  long distance = readDistance();

  // LDR
  int ldrValue = analogRead(LDR_PIN);

  // 0-1023 -> 0-100%
  int ldrPercent = map(
    ldrValue,
    0,
    1023,
    0,
    100
  );

  ldrPercent = constrain(
    ldrPercent,
    0,
    100
  );


  // Format:
  // FL,FR,RL,RR,DIST,SERVO,LDR

  Serial1.print(FL);
  Serial1.print(",");

  Serial1.print(FR);
  Serial1.print(",");

  Serial1.print(RL);
  Serial1.print(",");

  Serial1.print(RR);
  Serial1.print(",");

  Serial1.print(distance);
  Serial1.print(",");

  Serial1.print(servoPosition);
  Serial1.print(",");

  Serial1.println(ldrPercent);
}


// ==================== COMENZI ESP32 ====================

void readESPCommands() {

  while (Serial1.available()) {

    char c = Serial1.read();

    if (c == '\n') {

      espCommand.trim();


      if (espCommand == "MODE_MANUAL") {

        currentMode = MODE_MANUAL;

        motorStop();

        servoPosition = 90;
        servo.write(90);
      }


      else if (espCommand == "MODE_AVOIDANCE") {

        currentMode = MODE_AVOIDANCE;

        motorStop();

        servoPosition = servoCenter;
        servo.write(servoCenter);
      }


      else if (espCommand == "HORN") {

        buzzer();
      }


      else if (currentMode == MODE_MANUAL) {

        // ==================================================
        // MAPAREA MANUALA
        //
        // FAȚĂ    -> motorLeft()
        // SPATE   -> motorRight()
        // STÂNGA  -> manualLeft()
        // DREAPTA -> manualRight()
        // ==================================================

        if (espCommand == "MOVE_F") {

          motorLeft();
        }


        else if (espCommand == "MOVE_B") {

          motorRight();
        }


        else if (espCommand == "MOVE_L") {

          manualLeft();
        }


        else if (espCommand == "MOVE_R") {

          manualRight();
        }


        else if (espCommand == "MOVE_STOP") {

          motorStop();
        }
      }


      espCommand = "";
    }


    else {

      espCommand += c;

      if (espCommand.length() > 40) {
        espCommand = "";
      }
    }
  }
}


// ==================== SETUP ====================

void setup() {

  Serial.begin(115200);

  Serial1.begin(BAUDRATE);


  servo.attach(SERVO_PIN);

  servo.write(90);


  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  digitalWrite(TRIG_PIN, LOW);


  pinMode(IR_FRONT_LEFT, INPUT);
  pinMode(IR_FRONT_RIGHT, INPUT);
  pinMode(IR_REAR_LEFT, INPUT);
  pinMode(IR_REAR_RIGHT, INPUT);


  pinMode(ENA, OUTPUT);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(ENB, OUTPUT);


  pinMode(BUZZER_PIN, OUTPUT);


  pinMode(LDR_PIN, INPUT);


  pinMode(LED_FRONT_LEFT, OUTPUT);
  pinMode(LED_FRONT_RIGHT, OUTPUT);

  pinMode(LED_SIDE_LEFT, OUTPUT);
  pinMode(LED_SIDE_RIGHT, OUTPUT);

  pinMode(LED_REAR_LEFT, OUTPUT);
  pinMode(LED_REAR_RIGHT, OUTPUT);


  motorStop();

  updateLEDs();
}


// ==================== LOOP ====================

void loop() {

  updateIRParkingBuzzer();


  readESPCommands();


  updateLEDs();


  if (currentMode == MODE_AVOIDANCE) {

    runAvoidance();
  }


  static unsigned long lastSend = 0;

  if (millis() - lastSend >= 300) {

    lastSend = millis();

    sendSensorData();
  }


  delay(5);
}

void readMegaSensorDataTask(void *parameter) {

  String line = "";

  for (;;) {

    while (Serial.available()) {

      char c = Serial.read();

      if (c == '\n') {

        line.trim();

        // -------------------------------------------------
        // Cautam primele 6 virgule
        // FL,FR,RL,RR,DIST,SERVO,LDR
        // -------------------------------------------------

        int p1 = line.indexOf(',');
        int p2 = line.indexOf(',', p1 + 1);
        int p3 = line.indexOf(',', p2 + 1);
        int p4 = line.indexOf(',', p3 + 1);
        int p5 = line.indexOf(',', p4 + 1);
        int p6 = line.indexOf(',', p5 + 1);

        if (
          p1 > 0 &&
          p2 > p1 &&
          p3 > p2 &&
          p4 > p3
        ) {

          int fl = line.substring(0, p1).toInt();

          int fr = line.substring(
            p1 + 1,
            p2
          ).toInt();

          int rl = line.substring(
            p2 + 1,
            p3
          ).toInt();

          int rr = line.substring(
            p3 + 1,
            p4
          ).toInt();

          int dist = 0;
          int servo = 115;
          int ldr = 0;

          // DIST
          if (p5 > p4) {

            dist = line.substring(
              p4 + 1,
              p5
            ).toInt();

          } else {

            dist = line.substring(
              p4 + 1
            ).toInt();
          }

          // SERVO
          if (p6 > p5) {

            servo = line.substring(
              p5 + 1,
              p6
            ).toInt();

            // LDR
            ldr = line.substring(
              p6 + 1
            ).toInt();

          } else {

            servo = line.substring(
              p5 + 1
            ).toInt();
          }

          // -------------------------------------------------
          // VALIDARE IR
          // -------------------------------------------------

          if (
            (fl == 0 || fl == 1) &&
            (fr == 0 || fr == 1) &&
            (rl == 0 || rl == 1) &&
            (rr == 0 || rr == 1)
          ) {

            irFrontLeft  = fl;
            irFrontRight = fr;
            irRearLeft   = rl;
            irRearRight  = rr;

            hcDistance = constrain(
              dist,
              0,
              250
            );

            servoPosition = servo;

            ldrPercent = constrain(
              ldr,
              0,
              100
            );

            lastSensorFrame = millis();
          }
        }

        line = "";
      }

      else {

        if (c != '\r') {

          line += c;

          if (line.length() > 80) {
            line = "";
          }
        }
      }
    }

    vTaskDelay(
      pdMS_TO_TICKS(2)
    );
  }
}

void handleMode() {

  if (!server.hasArg("mode")) {

    server.send(
      400,
      "text/plain",
      "MODE lipseste"
    );

    return;
  }

  String mode =
    server.arg("mode");


  if (mode == "MANUAL") {

    Serial.println(
      "MODE_MANUAL"
    );

    server.send(
      200,
      "text/plain",
      "MANUAL"
    );

    return;
  }


  if (mode == "AVOIDANCE") {

    Serial.println(
      "MODE_AVOIDANCE"
    );

    server.send(
      200,
      "text/plain",
      "AVOIDANCE"
    );

    return;
  }


  server.send(
    400,
    "text/plain",
    "MODE invalid"
  );
}


// =====================================================
// COMENZI
// =====================================================

void handleCommand() {

  if (!server.hasArg("cmd")) {

    server.send(
      400,
      "text/plain",
      "CMD lipseste"
    );

    return;
  }

  String cmd =
    server.arg("cmd");


  if (
    cmd == "MOVE_F" ||
    cmd == "MOVE_B" ||
    cmd == "MOVE_L" ||
    cmd == "MOVE_R" ||
    cmd == "MOVE_STOP"
  ) {

    Serial.println(cmd);

    server.send(
      200,
      "text/plain",
      "OK"
    );

    return;
  }


  server.send(
    400,
    "text/plain",
    "CMD invalid"
  );
}

// =====================================================
// CAMERA STREAM
// =====================================================

static esp_err_t stream_handler(
  httpd_req_t *req
) {

  camera_fb_t *fb = NULL;

  esp_err_t res = ESP_OK;

  char part_buf[64];


  static const char*
  STREAM_CONTENT_TYPE =
    "multipart/x-mixed-replace;boundary=frame";


  static const char*
  STREAM_BOUNDARY =
    "\r\n--frame\r\n";


  static const char*
  STREAM_PART =
    "Content-Type: image/jpeg\r\n"
    "Content-Length: %u\r\n\r\n";


  res =
    httpd_resp_set_type(
      req,
      STREAM_CONTENT_TYPE
    );


  if (res != ESP_OK) {

    return res;
  }


  httpd_resp_set_hdr(
    req,
    "Access-Control-Allow-Origin",
    "*"
  );


  while (true) {

    fb =
      esp_camera_fb_get();


    if (!fb) {

      res = ESP_FAIL;

    }

    else {

      size_t hlen =
        snprintf(
          part_buf,
          64,
          STREAM_PART,
          fb->len
        );


      res =
        httpd_resp_send_chunk(
          req,
          STREAM_BOUNDARY,
          strlen(
            STREAM_BOUNDARY
          )
        );


      if (res == ESP_OK) {

        res =
          httpd_resp_send_chunk(
            req,
            part_buf,
            hlen
          );
      }


      if (res == ESP_OK) {

        res =
          httpd_resp_send_chunk(
            req,
            (const char*)fb->buf,
            fb->len
          );
      }


      esp_camera_fb_return(fb);
    }


    if (res != ESP_OK) {

      break;
    }
  }


  return res;
}


// =====================================================
// PORNIRE SERVER CAMERA
// =====================================================

void startCameraServer() {

  httpd_config_t config =
    HTTPD_DEFAULT_CONFIG();


  config.server_port = 81;

  config.ctrl_port = 32769;


  httpd_uri_t stream_uri;


  stream_uri.uri =
    "/stream";

  stream_uri.method =
    HTTP_GET;

  stream_uri.handler =
    stream_handler;

  stream_uri.user_ctx =
    NULL;


  if (
    httpd_start(
      &stream_httpd,
      &config
    ) == ESP_OK
  ) {

    httpd_register_uri_handler(
      stream_httpd,
      &stream_uri
    );
  }
}


// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(9600);


  // ---------------------------------------------------
  // CAMERA CONFIG
  // ---------------------------------------------------

  camera_config_t config;


  config.ledc_channel =
    LEDC_CHANNEL_0;

  config.ledc_timer =
    LEDC_TIMER_0;


  config.pin_d0 =
    Y2_GPIO_NUM;

  config.pin_d1 =
    Y3_GPIO_NUM;

  config.pin_d2 =
    Y4_GPIO_NUM;

  config.pin_d3 =
    Y5_GPIO_NUM;

  config.pin_d4 =
    Y6_GPIO_NUM;

  config.pin_d5 =
    Y7_GPIO_NUM;

  config.pin_d6 =
    Y8_GPIO_NUM;

  config.pin_d7 =
    Y9_GPIO_NUM;


  config.pin_xclk =
    XCLK_GPIO_NUM;

  config.pin_pclk =
    PCLK_GPIO_NUM;

  config.pin_vsync =
    VSYNC_GPIO_NUM;

  config.pin_href =
    HREF_GPIO_NUM;


  config.pin_sscb_sda =
    SIOD_GPIO_NUM;

  config.pin_sscb_scl =
    SIOC_GPIO_NUM;


  config.pin_pwdn =
    PWDN_GPIO_NUM;

  config.pin_reset =
    RESET_GPIO_NUM;


  config.xclk_freq_hz =
    20000000;


  config.pixel_format =
    PIXFORMAT_JPEG;


  if (psramFound()) {

    config.frame_size =
      FRAMESIZE_VGA;

    config.jpeg_quality =
      10;

    config.fb_count =
      2;

  }

  else {

    config.frame_size =
      FRAMESIZE_QVGA;

    config.jpeg_quality =
      12;

    config.fb_count =
      1;
  }


  // ---------------------------------------------------
  // INIT CAMERA
  // ---------------------------------------------------

  if (
    esp_camera_init(
      &config
    ) != ESP_OK
  ) {

    while (true) {

      delay(1000);
    }
  }


  // ---------------------------------------------------
  // WIFI
  // ---------------------------------------------------

  WiFi.begin(
    ssid,
    password
  );


  while (
    WiFi.status() !=
    WL_CONNECTED
  ) {

    delay(500);
  }


  // ---------------------------------------------------
  // WEB SERVER
  // ---------------------------------------------------

  server.on(
    "/",
    HTTP_GET,
    handleRoot
  );


  server.on(
    "/mode",
    HTTP_GET,
    handleMode
  );


  server.on(
    "/command",
    HTTP_GET,
    handleCommand
  );


  server.on(
    "/sensors",
    HTTP_GET,
    handleSensors
  );


  server.on(
    "/horn",
    HTTP_GET,
    handleHorn
  );


  server.begin();


  // ---------------------------------------------------
  // CAMERA SERVER PORT 81
  // ---------------------------------------------------

  startCameraServer();


  // ---------------------------------------------------
  // UART TASK
  // ---------------------------------------------------

  xTaskCreatePinnedToCore(

    readMegaSensorDataTask,

    "UART_IR",

    4096,

    NULL,

    2,

    &uartTaskHandle,

    1
  );
}
